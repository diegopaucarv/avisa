function doLogin() {
  const dni   = document.getElementById('lDNI').value.trim();
  const clave = document.getElementById('lClave').value.trim();

  if (!dni || !clave) {
    mostrarError('Por favor ingrese su DNI y clave');
    return;
  }
  if (dni.length < 8) {
    mostrarError('El DNI debe tener 8 d\\u00EDgitos');
    return;
  }

  // Mostrar loader
  document.getElementById('btnLogin').disabled = true;
  document.getElementById('lload').style.display = 'block';
  document.getElementById('lerr').style.display  = 'none';

  // Llamar a Apps Script
  google.script.run
    .withSuccessHandler(resultado => {
      document.getElementById('btnLogin').disabled = false;
      document.getElementById('lload').style.display = 'none';

      if (resultado && resultado.ok) {
        // Guardar sesi\\u00F3n
        SESSION = {
          dni:    dni,
          nombre: resultado.nombre,
          rol:    resultado.rol,
          id:     resultado.id,
          area:   resultado.area   || '',
          email:  resultado.email  || '',
          cargo:  resultado.cargo  || ''
        };
        iniciarApp();
      } else {
        mostrarError(resultado.msg || 'DNI o clave incorrectos');
        document.getElementById('lDNI').classList.add('err');
        document.getElementById('lClave').classList.add('err');
      }
    })
    .withFailureHandler(err => {
      document.getElementById('btnLogin').disabled = false;
      document.getElementById('lload').style.display = 'none';
      mostrarError('Error de conexi\\u00F3n. Intente nuevamente.');
      console.error('Error login:', err);
    })
    .verificarLogin(dni, clave);
}

function iniciarApp() {
  // Ocultar login, mostrar app
  document.getElementById('loginScreen').style.display = 'none';
  document.getElementById('app').style.display = 'block';

  // Actualizar header
  const ini = SESSION.nombre.split(' ').map(w=>w[0]).slice(0,2).join('').toUpperCase();
  document.getElementById('hAv').textContent   = ini;
  document.getElementById('hName').textContent = SESSION.nombre.split(' ').slice(0,2).join(' ');
  document.getElementById('hRole').textContent = SESSION.rol;
  document.getElementById('hDate').textContent =
    new Date().toLocaleDateString('es-PE',{day:'2-digit',month:'short',year:'numeric'});
  document.getElementById('dashBienvenida').textContent =
    'Bienvenido/a, ' + SESSION.nombre.split(' ')[0] + ' \\u2014 ' + SESSION.cargo;

  // Control de visibilidad por rol
  aplicarPermisosPorRol();
  mostrarBienvenida(SESSION.nombre.split(' ')[0]);

  // Si es PF o Autoridad ir directo a Pendientes
  if (SESSION.rol === 'PuntoFocal' || SESSION.rol === 'Autoridad') {
    goM('pendientes', document.querySelector('[onclick*=pendientes]'));
  } else {
    cargarDashboard();
    // Pre-cargar pendientes en background para badge
    google.script.run
      .withSuccessHandler(res => {
        const nb = document.getElementById('nbPendientes');
        if (nb && res.ok) nb.textContent = res.data.length;
      })
      .withFailureHandler(() => {})
      .getPendientes(SESSION.id, SESSION.rol);
  }
}

// PERMISOS POR ROL
// \\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550\\u2550
function aplicarPermisosPorRol() {
  const rol = SESSION.rol;

  // \\u2500\\u2500 Mostrar/ocultar sidebar seg\\u00FAn rol \\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500
  // PuntoFocal y Autoridad tienen vista limitada
  if (rol === 'PuntoFocal') {
    // Ocultar TODO el sidebar menos Pendientes
    document.querySelectorAll('.nav').forEach(n => {
      const txt = n.textContent.trim();
      if (!txt.includes('Pendientes')) n.style.display = 'none';
    });
    document.querySelectorAll('.sb-sec').forEach(s => s.style.display = 'none');
  }

  if (rol === 'Autoridad') {
    // Solo Dashboard principal y Pendientes
    document.querySelectorAll('.nav').forEach(n => {
      const txt = n.textContent.trim();
      if (!txt.includes('Dashboard') && !txt.includes('Pendientes')) {
        n.style.display = 'none';
      }
    });
    document.querySelectorAll('.sb-sec').forEach(s => s.style.display = 'none');
  }

  // Admin \\u2014 ver todo incluyendo Maestros y Log
  if (rol === 'Admin') {
    document.querySelectorAll('.role-admin-only')
            .forEach(el => el.style.display = 'flex');
    document.getElementById('sbAdminSec').style.display = 'block';
  }

  // Directora \\u2014 ver Log de Env\\u00EDos tambi\\u00E9n
  if (rol === 'Directora') {
    const sbLog = document.getElementById('sbLog');
    if (sbLog) sbLog.style.display = 'flex';
  }
}

// \\u2500\\u2500 Verificar si puede validar un tipo \\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500\\u2500
function puedeValidar(tipo) {
  const rol = SESSION.rol;
  if (rol === 'Admin') return true;
  if (tipo === 'caso'    || tipo === 'alerta')    return rol === 'Directora' || rol === 'PuntoFocal';
  if (tipo === 'accion')                          return rol === 'Directora' || rol === 'PuntoFocal';
  if (tipo === 'reporte' || tipo === 'clima')     return rol === 'Directora';
  return false;
}

  if (!contenedor) return;

  if (!puedeDescargar()) {
    contenedor.querySelectorAll('.bpdf, .bxl').forEach(b => b.style.display = 'none');
  }
  // Botones de crear
  if (!puedeCrear()) {
    contenedor.querySelectorAll('.btn-crear').forEach(b => b.style.display = 'none');
  }

function goM(id, el) {
  document.querySelectorAll('.mod').forEach(m => m.classList.remove('on'));
  const mod = document.getElementById('mod-' + id);
  if (mod) mod.classList.add('on');
  document.querySelectorAll('.nav').forEach(n => n.classList.remove('on'));
  if (el) el.classList.add('on');
  if (window.innerWidth <= 768) document.getElementById('sidebar').classList.remove('open');

  // Cargar datos del m\\u00F3dulo si a\\u00FAn no est\\u00E1n en cache
  switch(id) {
    case 'pendientes': cargarPendientes(); break;
    case 'casos':      if (!CACHE.casos)      cargarCasos();      break;
    case 'alertas':    if (!CACHE.alertas)    cargarAlertas();    break;
    case 'rr':         if (!CACHE.rr)         cargarRR();         break;
    case 'clima':      if (!CACHE.clima)      cargarClima();      break;
    case 'hechos':     if (!CACHE.hechos)     cargarHechos();     break;
    case 'acciones':   if (!CACHE.acciones)   cargarAcciones();   break;
    case 'actores':    if (!CACHE.actores)    cargarActores();    break;
    case 'sectoristas':if (!CACHE.sectoristas)cargarSectoristas();break;
    case 'pf':         if (!CACHE.pf)         cargarPF();         break;
    case 'autoridades':if (!CACHE.autoridades)cargarAutoridades();break;
    case 'maestros':   cargarMaestros();                          break;
    case 'log':        cargarLog();                               break;
  }
}

function sTab(prefix, id, btn) {
  const mod = document.getElementById('mod-' + prefix) ||
              document.getElementById(prefix + 'Content');
  if (mod) {
    mod.querySelectorAll('.tpane').forEach(t => t.classList.remove('on'));
    const p = document.getElementById(prefix + '-' + id);
    if (p) p.classList.add('on');
  }
  if (btn) {
    btn.closest('.tabs').querySelectorAll('.tb').forEach(b => b.classList.remove('on'));
    btn.classList.add('on');
  }
}

function fmtFecha(f) {

// Excel export
function exportCSV(tableId, filename) {
  const tbl = document.getElementById(tableId);
  if (!tbl) return;
  let csv = '';
  tbl.querySelectorAll('tr').forEach(row => {
    const vals = [...row.querySelectorAll('th,td')]
      .map(c => '"' + c.innerText.replace(/"/g,'""') + '"');
    csv += vals.join(',') + '\\\\n';
  });
  const blob = new Blob(['\\\\uFEFF' + csv], {type:'text/csv;charset=utf-8'});
  const url  = URL.createObjectURL(blob);
  const a    = document.createElement('a');
  a.href = url; a.download = filename + '_' + new Date().toISOString().slice(0,10) + '.csv';
  document.body.appendChild(a); a.click(); document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

// B\\u00FAsqueda global
function buscarGlobal(q) {
  if (!q || q.length < 2) return;
  q = q.toLowerCase();
  if (CACHE.casos) {
    const m = CACHE.casos.find(c =>
      (c.Codigo_Caso||'').toLowerCase().includes(q) ||
      (c.Nombre_Caso||'').toLowerCase().includes(q));
    if (m) { goM('casos', document.querySelector('[onclick*=casos]')); return; }
  }
  if (CACHE.alertas) {
    const m = CACHE.alertas.find(a =>
      (a.ID_Alerta||'').toLowerCase().includes(q) ||
      (a.Titulo_Alerta||'').toLowerCase().includes(q));
    if (m) { goM('alertas', document.querySelector('[onclick*=alertas]')); return; }
  }

function cargarDashboard() {
  document.getElementById('dashLoading').style.display = 'block';
  document.getElementById('dashContent').style.display = 'none';

  // Cargar casos y alertas en paralelo
  Promise.all([
    new Promise((res,rej) => {
      google.script.run
        .withSuccessHandler(d => res(d))
        .withFailureHandler(e => rej(e))
        .getCasos();
    }),
    new Promise((res,rej) => {
      google.script.run
        .withSuccessHandler(d => res(d))
        .withFailureHandler(e => rej(e))
        .getAlertas();
    })
  ]).then(([resCasos, resAlertas]) => {
    if (resCasos && resCasos.ok)   CACHE.casos   = resCasos.data;
    if (resAlertas && resAlertas.ok) CACHE.alertas = resAlertas.data;

  }).catch(err => {
    console.error('Error cargando dashboard:', err);
    // Mostrar igualmente con datos vac\\u00EDos
    renderDashboard();
  });
}

function cargarCasos() {
  document.getElementById('casosLoading').style.display = 'block';
  document.getElementById('casosContent').style.display = 'none';

  // Cargar datos maestros en paralelo si no est\\u00E1n en cache
  const promesas = [
    CACHE.casos ? Promise.resolve({ok:true,data:CACHE.casos})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getCasos()),
    CACHE.tipoCaso ? Promise.resolve({ok:true,data:CACHE.tipoCaso})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getTipoCaso()),
    CACHE.sectores ? Promise.resolve({ok:true,data:CACHE.sectores})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getSectores()),
    CACHE.tematica ? Promise.resolve({ok:true,data:CACHE.tematica})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getTematica()),
    CACHE.sectoristas ? Promise.resolve({ok:true,data:CACHE.sectoristas})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getSectoristas()),
    CACHE.departamentos ? Promise.resolve({ok:true,data:CACHE.departamentos})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getDepartamentos()),
    CACHE.intervencion ? Promise.resolve({ok:true,data:CACHE.intervencion})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getIntervencion()),
    CACHE.pasosCaso ? Promise.resolve({ok:true,data:CACHE.pasosCaso})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getPasosCaso()),
  ];

  Promise.all(promesas).then(([
    resCasos, resTipo, resSect, resTema,
    resSec, resDepto, resInterv, resPasos
  ]) => {
    if (resCasos && resCasos.ok)   CACHE.casos        = resCasos.data;
    if (resTipo && resTipo.ok)    CACHE.tipoCaso     = resTipo.data;
    if (resSect && resSect.ok)    CACHE.sectores     = resSect.data;
    if (resTema && resTema.ok)    CACHE.tematica     = resTema.data;
    if (resSec && resSec.ok)     CACHE.sectoristas  = resSec.data;
    if (resDepto && resDepto.ok)   CACHE.departamentos= resDepto.data;
    if (resInterv && resInterv.ok)  CACHE.intervencion = resInterv.data;
    if (resPasos && resPasos.ok)   CACHE.pasosCaso    = resPasos.data;

    renderCasosModule();
  }).catch(err => {
    console.error('Error cargando m\\u00F3dulo casos:', err);
    renderCasosModule();
  });
}

        .withSuccessHandler(function(r){ if(r&&r.ok){ CACHE.departamentos=r.data; llenarSelect('nc_depto_add', r.data, 'ID_Ubigeo', 'Departamento'); } })

        .getDepartamentos();
    }
  }
  llenarBridgeDeptos();
  
  // Llenar bridge: Actores
  function llenarBridgeActores() {
    if (CACHE.actores && CACHE.actores.length) {
      llenarSelect('nc_actor_add', CACHE.actores, 'ID_Actor', 'Nombre_Actor');
    } else {
      google.script.run
        .withSuccessHandler(function(r){ if(r&&r.ok){ CACHE.actores=r.data; llenarSelect('nc_actor_add', r.data, 'ID_Actor', 'Nombre_Actor'); } })
        .withFailureHandler(function(){})
        .getActores();
    }
  }
  llenarBridgeActores();
  
  // Llenar bridge: Recursos pesqueros
  function llenarBridgeRecursos() {
    if (CACHE.recursos && CACHE.recursos.length) {
      var colR = 'Nombre_comun_Recurso';
      if (CACHE.recursos.length > 0) {
        var rKeys = Object.keys(CACHE.recursos[0]);
        var rFound = rKeys.find(function(k){ return (k.indexOf('Nombre_com') >= 0 || k.indexOf('Nombre_c') >= 0) && k.toLowerCase().indexOf('recurso') >= 0; });
        if (rFound) colR = rFound;
      }
      llenarSelect('nc_recurso_add', CACHE.recursos, 'ID_Recurso', colR);
    } else {
      google.script.run
        .withSuccessHandler(function(r){ 
        if(r&&r.ok){ 
          CACHE.recursos=r.data; 
          // Detectar el nombre de columna correcto (puede tener tilde)
          var colNombre = 'Nombre_comun_Recurso';
          if (r.data.length > 0) {
            var keys = Object.keys(r.data[0]);
            var found = keys.find(function(k){ return k.toLowerCase().replace(/[^a-z_]/g,'').indexOf('nombre') >= 0 && k.toLowerCase().indexOf('recurso') >= 0; });

        .getRecursos();
    }

  window._bridgeDeptos   = [];
  window._bridgeActores  = [];
  window._bridgeRecursos = [];
  var dLista = document.getElementById('nc_deptos_lista');
  var aLista = document.getElementById('nc_actores_lista');
  var rLista = document.getElementById('nc_recursos_lista');

    });
  }

  filtrarCasos();

  document.getElementById('casosLoading').style.display = 'none';
  document.getElementById('casosContent').style.display = 'block';
}

function filtrarCasos() {
  const q     = (document.getElementById('fCasoQ')?.value||'').toLowerCase();
  const prio  = document.getElementById('fCasoPrio')?.value||'';
  const riesgo= document.getElementById('fCasoRiesgo')?.value||'';
  const sector= document.getElementById('fCasoSector')?.value||'';
  let lista = CACHE.casos || [];

  // Filtro por rol: Sectorista solo ve sus casos
  if (SESSION.rol === 'Sectorista') {
    lista = lista.filter(function(c) {
      return c.ID_Sectorista === SESSION.id ||
             c.ID_Sectorista === SESSION.dni;
    });
  }
  // Sin columna Estado en el Sheets - mostrar todos los casos del sectorista
  // Los cerrados tienen Fecha_Cierre no vacía

  if (q) lista = lista.filter(c =>
    (c.Codigo_Caso||'').toLowerCase().includes(q) ||
    (c.Nombre_Caso||'').toLowerCase().includes(q));
  if (prio)   lista = lista.filter(c => c.Prioridad === prio);
  if (riesgo) lista = lista.filter(c => (c.Nivel_Riesgo_Actual||'').includes(riesgo));
  if (sector) lista = lista.filter(c => c.ID_Sector === sector);

  if (cnt) cnt.textContent = lista.length + ' registros';

  const tbody = document.querySelector('#tblCasos tbody');

  tbody.innerHTML = lista.length ? lista.map(c => \`<tr>

    <td class="trunc" title="\${c.Nombre_Caso||''}" style="max-width:190px">\${c.Nombre_Caso||''}</td>

function guardarNuevoCaso(tipo) {
  var getVal = function(id) { var el=document.getElementById(id); return el?el.value:''; };
  const datos = {
    Codigo_Caso:        getVal('nc_codigo'),
    ID_Tipo:            getVal('nc_tipo'),
    Nombre_Caso:        getVal('nc_nombre'),
    Resumen_Ejecutivo:  getVal('nc_resumen'),
    ID_Sector:          getVal('nc_sector'),
    ID_Tematica:        getVal('nc_tematica'),
    Fecha_Inicio:       getVal('nc_fecha'),
    ID_Departamento:    getVal('nc_depto'),
    ID_Provincia:       getVal('nc_prov'),
    ID_Distrito:        getVal('nc_dist'),
    ID_Sectorista:      getVal('nc_sectorista'),
    Prioridad:          getVal('nc_prioridad'),
    Nivel_Riesgo_Actual: getVal('nc_idRiesgo'),
    ID_Riesgos:         getVal('nc_idRiesgo'),
    ID_Intervencion:    getVal('nc_interv'),
    Descripcion_Riesgo: getVal('nc_descRiesgo'),
    _deptos:   window._bridgeDeptos   || [],
    _actores:  window._bridgeActores  || [],
    _recursos: window._bridgeRecursos || [],
  };

  if (!datos.Nombre_Caso || !datos.ID_Sectorista) {
    toast('Complete los campos obligatorios (*)','err'); return;
  }

  var btn = event.target;
  btn.disabled = true;
  btn.textContent = '\\u23F3 Guardando...';

  if (tipo === 'borrador') {
    // Guardar solo localmente en sessionStorage
    var borradores = JSON.parse(sessionStorage.getItem('borradores_casos') || '[]');
    // Generar ID temporal si no tiene
    if (!datos.Codigo_Caso) {
      datos.Codigo_Caso = 'BORR-' + Date.now();
    }
    datos.Estado = 'Borrador';
    datos._borrador = true;
    datos._ts = new Date().toISOString();
    // Actualizar o agregar
    var idx = borradores.findIndex(function(b){ return b.Codigo_Caso === datos.Codigo_Caso; });
    if (idx >= 0) borradores[idx] = datos;
    else borradores.push(datos);
    sessionStorage.setItem('borradores_casos', JSON.stringify(borradores));
    btn.disabled = false;
    btn.textContent = '\\uD83D\\uDCBE Guardar Borrador';
    toast('Borrador guardado. Use Enviar a Validaci\\u00F3n para registrar oficialmente.');
    // Mostrar en lista de pendientes del sectorista
    actualizarPendientesBorradores();
    return;
  }

  // Enviar a validaci\\u00F3n - grabar en Sheets
  datos.Estado = 'En revisi\\u00F3n';
  // Eliminar borrador local si exist\\u00EDa
  try {
    var borradores2 = JSON.parse(sessionStorage.getItem('borradores_casos') || '[]');
    sessionStorage.setItem('borradores_casos', JSON.stringify(
      borradores2.filter(function(b){ return b.Codigo_Caso !== datos.Codigo_Caso; })
    ));
  } catch(e) {}

  google.script.run
    .withSuccessHandler(function(res) {
      btn.disabled = false;
      btn.textContent = '\\uD83D\\uDCE4 Enviar a Validaci\\u00F3n';
      if (res.ok) {
        toast('Caso enviado a validaci\\u00F3n correctamente');
        // Guardar bridges si hay datos
        var codigoCaso = res.codigo || datos.Codigo_Caso;
        if (datos._deptos && datos._deptos.length > 0) {
          google.script.run.withSuccessHandler(function(){}).withFailureHandler(function(){})
            .guardarBridgeDeptos(codigoCaso, datos._deptos);
        }
        if (datos._actores && datos._actores.length > 0) {
          google.script.run.withSuccessHandler(function(){}).withFailureHandler(function(){})
            .guardarBridgeActores(codigoCaso, datos._actores);
        }
        if (datos._recursos && datos._recursos.length > 0) {
          google.script.run.withSuccessHandler(function(){}).withFailureHandler(function(){})
            .guardarBridgeRecursos(codigoCaso, datos._recursos);
        }
        CACHE.casos = null;
        cargarCasos();
        sTab('casos','lista',document.querySelector('#casosTabs .tb'));
      } else {
        toast(res.msg || 'Error al guardar','err');
      }
    })
    .withFailureHandler(function(err) {
      btn.disabled = false;
      toast('Error de conexi\\u00F3n','err');
    })
    .guardarCaso(datos);
}

    var nb = document.getElementById('nbPendientes');

  } catch(e) {}
}

function avanzarPasoCaso(idCaso, codigoCaso, pasoActual, accion) {
  const obs = document.getElementById('obsFlujoCaso')?.value || '';
  if (accion === 'devolver' && !obs.trim()) {
    toast('Escriba una observaci\\u00F3n para devolver el caso','err'); return;
  }

  google.script.run
    .withSuccessHandler(res => {
      if (res.ok) {
        toast(accion === 'aprobar' ? '\\u2705 Caso avanzado al siguiente paso' : '\\u21A9\\uFE0F Caso devuelto con observaci\\u00F3n');
        CACHE.casos = null;
        cerrarModal();
        cargarCasos();
      } else {
        toast(res.msg,'err');
      }
    })
    .withFailureHandler(() => toast('Error de conexi\\u00F3n','err'))
    .avanzarPaso(idCaso, pasoActual, SESSION.id, obs, accion);
}

function generarPDFCaso(idCaso, codigoCaso) {
  toast('\\u23F3 Generando PDF...');
  const htmlContenido = document.getElementById('modalDetalleInner')?.innerHTML || '';
  const destinatarios = obtenerDestinatariosCaso(idCaso);

  google.script.run
    .withSuccessHandler(res => {
      if (res.ok && res.resultados?.pdf?.url) {
        window.open(res.resultados.pdf.url, '_blank');
        toast('\\u2705 PDF generado y disponible en Drive');
        // Mostrar bot\\u00F3n WhatsApp con link real
        const btnWA = document.getElementById('btnWACaso');
        if (btnWA && res.resultados.whatsapp?.length) {
          btnWA.onclick = () => window.open(res.resultados.whatsapp[0].link, '_blank');
        }
      } else {
        toast(res.msg || 'Error al generar PDF','err');
      }
    })
    .withFailureHandler(() => toast('Error generando PDF','err'))
    .enviarDocumento('caso', codigoCaso, htmlContenido, destinatarios);
}

function enviarCorreoCaso(idCaso, codigoCaso) {
  const destinatarios = obtenerDestinatariosCaso(idCaso);
  if (!destinatarios.emails.filter(Boolean).length) {
    toast('No hay destinatarios con email configurado','err'); return;
  }
  toast('\\u23F3 Enviando correo...');
  const htmlContenido = document.getElementById('modalDetalleInner')?.innerHTML || '';

  google.script.run
    .withSuccessHandler(res => {
      if (res.ok) toast('\\u2705 Correo enviado correctamente');
      else toast(res.msg,'err');
    })
    .withFailureHandler(() => toast('Error enviando correo','err'))
    .enviarDocumento('caso', codigoCaso, htmlContenido, destinatarios);
}

function obtenerDestinatariosCaso(idCaso) {
  // Buscar directora en autoridades
  const aut = (CACHE.autoridades||[]).find(a =>
    (a.Cargo||'').toLowerCase().includes('director'));
  const emails    = [aut?.Email||''].filter(Boolean);
  const telefonos = [aut?.Telefono||''];
  const nombres   = [aut?.Nombre_Autoridades||'Directora'];
  return { emails, telefonos, nombres };
}

function cargarAlertas() {
  document.getElementById('alertasLoading').style.display = 'block';
  document.getElementById('alertasContent').style.display = 'none';

  const promesas = [
    CACHE.alertas ? Promise.resolve({ok:true,data:CACHE.alertas})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getAlertas()),
    CACHE.pasosAlerta ? Promise.resolve({ok:true,data:CACHE.pasosAlerta})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getPasosAlerta()),
    CACHE.sectoristas ? Promise.resolve({ok:true,data:CACHE.sectoristas})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getSectoristas()),
    CACHE.departamentos ? Promise.resolve({ok:true,data:CACHE.departamentos})
      : new Promise((r,j) => google.script.run
          .withSuccessHandler(r).withFailureHandler(j).getDepartamentos()),
  ];

  Promise.all(promesas).then(([resAl, resPasos, resSec, resDepto]) => {
    if (resAl && resAl.ok)    CACHE.alertas      = resAl.data;
    if (resPasos && resPasos.ok) CACHE.pasosAlerta  = resPasos.data;
    if (resSec && resSec.ok)   CACHE.sectoristas  = resSec.data;
    if (resDepto && resDepto.ok) CACHE.departamentos= resDepto.data;
    renderAlertasModule();
  }).catch(err => { console.error(err); renderAlertasModule(); });
}

function filtrarAlertas() {
  const q     = (document.getElementById('fAlertaQ')?.value||'').toLowerCase();
  const est   = document.getElementById('fAlertaEst')?.value||'';
  const sector= document.getElementById('fAlertaSec')?.value||'';
  let lista = CACHE.alertas || [];

  if (SESSION.rol === 'Sectorista') {
    lista = lista.filter(a => a.Sectorista === SESSION.id);
  }
  if (q)      lista = lista.filter(a =>
    (a.Titulo_Alerta||'').toLowerCase().includes(q) ||
    (a.ID_Alerta||'').toLowerCase().includes(q));
  if (est)    lista = lista.filter(a => a.Estado === est);
  if (sector) lista = lista.filter(a => a.ID_Sector === sector);

  const cnt = document.getElementById('cntAlertas');
  if (cnt) cnt.textContent = lista.length + ' registros';

  const tbody = document.querySelector('#tblAlertas tbody');
  if (!tbody) return;

  tbody.innerHTML = lista.length ? lista.map(a => {
    const riesgoLabel = a.ID_Riesgos==='RIE-001'?'Alto':a.ID_Riesgos==='RIE-002'?'Medio':'Bajo';
    const riesgoBadge = a.ID_Riesgos==='RIE-001'?'b-red':a.ID_Riesgos==='RIE-002'?'b-amb':'b-grn';
    return \`<tr>
      <td><code>\${a.ID_Alerta||''}</code></td>
      <td class="trunc" title="\${a.Titulo_Alerta||''}" style="max-width:200px">\${a.Titulo_Alerta||''}</td>
      <td style="font-size:12px">\${fmtFecha(a.Fecha)}</td>
      <td>\${a.ID_Provincia||a.ID_Ubigeo||'\\u2014'}</td>
      <td style="font-size:12px">\${sLabel(a.ID_Sector)}</td>
      <td><span class="badge \${riesgoBadge}">\${riesgoLabel}</span></td>
      <td>\${sBadge(a.Estado)}</td>
      <td>\${a.ID_Caso_Vinculado?\`<code>\${a.ID_Caso_Vinculado}</code>\`:'\\u2014'}</td>
      <td>
        <div class="t-act">
          <button class="btn-ico" title="Ver detalle" onclick='abrirDetalleAlerta(\${JSON.stringify(a)})'>\\uD83D\\uDC41\\uFE0F</button>
          <button class="btn-ico" title="Editar">\\u270F\\uFE0F</button>

        </div>
      </td>
    </tr>\`;
  }).join('')

// WhatsApp con formato igual a la imagen del sectorista
function generarMensajeWA(tipo, doc) {
  const tipos = {
    alerta:  '\\uD83D\\uDD14 REPORTE DE ALERTA TEMPRANA',
    rr:      '\\uD83D\\uDCC4 REPORTE R\\u00C1PIDO DE INTERVENCI\\u00D3N',
    clima:   '\\uD83C\\uDF24\\uFE0F REPORTE DE CLIMA SOCIAL',
    caso:    '\\uD83D\\uDCC1 CASO DE CONFLICTO SOCIAL'
  };
  const encabezado = tipos[tipo] || '\\uD83D\\uDCCB DOCUMENTO SIPRECS';
  const titulo = doc.Titulo_Alerta || doc.Titulo_Reporte_Rapido ||
                 doc.Titulo_Clima  || doc.Nombre_Caso || '';
  const fecha  = doc.Fecha ? fmtFecha(doc.Fecha) : '';
  const ubigeo = doc.ID_Provincia || doc.ID_Ubigeo || '';
  const sitActual = doc.Situaci\\u00F3n_Actual || doc.Resumen_Ejecutivo ||
                    doc.Situaci\\u00F3n_Actual_Clima || '';

  let msg = \`\\uD83D\\uDEE1\\uFE0F *SIPRECS \\u2014 PRODUCE*\\\\n\`;
  msg += \`_Oficina de Articulaci\\u00F3n Intergubernamental_\\\\n\\\\n\`;
  msg += \`\${encabezado}\\\\n\`;
  msg += \`_\${titulo}_\\\\n\\\\n\`;
  if (fecha)  msg += \`\\uD83D\\uDCC5 Fecha de emisi\\u00F3n: \${fecha}\`;
  if (ubigeo) msg += \` | \\uD83D\\uDCCD \\u00C1mbito: \${ubigeo}\`;
  if (fecha || ubigeo) msg += '\\\\n\\\\n';
  if (sitActual) {
    msg += \`\\uD83D\\uDD0D *SITUACI\\u00D3N ACTUAL*\\\\n\${sitActual}\\\\n\\\\n\`;
  }
  if (doc.Link_PDF_Drive) {
    msg += \`\\uD83D\\uDCC4 *Ver documento completo:*\\\\n\${doc.Link_PDF_Drive}\\\\n\\\\n\`;
  }
  msg += \`_Ingrese al sistema con su DNI y clave para m\\u00E1s detalles._\`;
  return msg;
}

function abrirWhatsAppAlerta(idAlerta) {
  const alerta = (CACHE.alertas||[]).find(a => a.ID_Alerta === idAlerta);
  if (!alerta) { toast('Alerta no encontrada','err'); return; }

  // Buscar tel\\u00E9fono de la directora
  const aut = (CACHE.autoridades||[]).find(a =>
    (a.Cargo||'').toLowerCase().includes('director'));
  const tel = aut?.Telefono||'';

  if (!tel) {
    // Sin tel\\u00E9fono configurado: mostrar el mensaje para copiar
    const msg = generarMensajeWA('alerta', alerta);
    const encoded = encodeURIComponent(msg);
    window.open(\`https://wa.me/?text=\${encoded}\`, '_blank');
  } else {
    const msg = generarMensajeWA('alerta', alerta);
    const telFull = tel.startsWith('51') ? tel : '51' + tel.replace(/\\\\D/g,'');
    const encoded = encodeURIComponent(msg);
    // Registrar clic en LOG
    registrarEnvioWA('alerta', idAlerta);
    window.open(\`https://wa.me/\${telFull}?text=\${encoded}\`, '_blank');
  }
}

function registrarEnvioWA(tipo, idDoc) {
  google.script.run
    .withSuccessHandler(() => {})
    .withFailureHandler(() => {})
    .registrarEnvioLog({
      Canal: 'WhatsApp',
      ID_Tipo_Reporte: tipo === 'alerta' ? 'TR-01' :
                       tipo === 'clima'  ? 'TR-02' :
                       tipo === 'rr'     ? 'TR-03' : 'TR-04',
      ID_Documento:    idDoc,
      Paso_Num:        1,

function abrirWhatsAppClima(idClima) {
  const c=(CACHE.clima||[]).find(x=>x.ID_Clima===idClima);

  const msg=generarMensajeWA('clima',c);
  const aut=(CACHE.autoridades||[]).find(a=>(a.Cargo||'').toLowerCase().includes('director'));
  const tel=(aut?.Telefono||'').replace(/\\\\D/g,'');
  const telFull=tel?('51'+tel):'';
  const url=telFull?\`https://wa.me/\${telFull}?text=\${encodeURIComponent(msg)}\`:\`https://wa.me/?text=\${encodeURIComponent(msg)}\`;
  registrarEnvioWA('clima',idClima);
  window.open(url,'_blank');
}

function cargarProvincias(selId, deptoId) {
  const sel = document.getElementById(selId);
  if (!sel) return;
  sel.innerHTML = '<option value="">-- Provincia --</option>';
  (CACHE.provincias||[]).filter(p => p.ID_Ubigeo === deptoId).forEach(p => {
    const o = document.createElement('option');
    o.value = p.ID_Provincia;
    o.textContent = p.Nombre_Provincia;
    sel.appendChild(o);
  });
  // Si no hay provincias en cache, cargar desde Apps Script
  if (!CACHE.provincias) {
    google.script.run
      .withSuccessHandler(res => {
        if (res.ok) {
          CACHE.provincias = res.data;
          cargarProvincias(selId, deptoId);
        }
      })
      .withFailureHandler(() => {})
      .getProvincias();
  }
}

function previsualizarFoto(input, prevId, dropId) {
  const file = input.files[0];
  if (!file) return;
  if (file.size > 4 * 1024 * 1024) {
    toast('La foto debe pesar menos de 4MB', 'err'); return;
  }
  const reader = new FileReader();
  reader.onload = e => {
    const prev = document.getElementById(prevId);
    if (prev) { prev.src = e.target.result; prev.style.display = 'block'; }
  };
  reader.readAsDataURL(file);
}

    CACHE.hechos ? Promise.resolve({ok:true,data:CACHE.hechos})
      : new Promise((r,j)=>google.script.run.withSuccessHandler(r).withFailureHandler(j).getHechos()),
    CACHE.casos ? Promise.resolve({ok:true,data:CACHE.casos})
      : new Promise((r,j)=>google.script.run.withSuccessHandler(r).withFailureHandler(j).getCasos()),

    if(resC.ok)  CACHE.casos       = resC.data;
    if(resSec.ok)CACHE.sectoristas = resSec.data;

  }).catch(err=>{console.error(err);renderHechosModule();});
}

function cargarAcciones() {
  const promesas = [
    CACHE.acciones ? Promise.resolve({ok:true,data:CACHE.acciones})
      : new Promise((r,j)=>google.script.run.withSuccessHandler(r).withFailureHandler(j).getAccionesHitos()),
    CACHE.casos ? Promise.resolve({ok:true,data:CACHE.casos})
      : new Promise((r,j)=>google.script.run.withSuccessHandler(r).withFailureHandler(j).getCasos()),
    CACHE.pf ? Promise.resolve({ok:true,data:CACHE.pf})
      : new Promise((r,j)=>google.script.run.withSuccessHandler(r).withFailureHandler(j).getPuntosFocales()),
  ];
  Promise.all(promesas).then(([resA, resC, resPF])=>{
    if(resA.ok) CACHE.acciones = resA.data;
    if(resC.ok) CACHE.casos    = resC.data;
    if(resPF.ok)CACHE.pf       = resPF.data;
    renderAccionesModule();
  }).catch(err=>{console.error(err);renderAccionesModule();});
}

function guardarActor() {
  const nombre = document.getElementById('act_nombre').value.trim();
  const cargo  = document.getElementById('act_cargo').value.trim();
  if (!nombre || !cargo) {
    toast('Complete nombre y cargo (*)','err'); return;
  }
  const datos = {
    Nombre_Actor:           nombre,
    DNI:                    document.getElementById('act_dni').value,
    Telefono:               document.getElementById('act_tel').value,
    Cargo:                  cargo,
    Instituci\\u00F3n_Organizaci\\u00F3n: document.getElementById('act_org').value,
    ID_Tipo_Actor:          document.getElementById('act_tipo').value,
    Nivel_Gobierno:         document.getElementById('act_nivel').value,
    Categoria:              document.getElementById('act_cat').value,
    Activo:                 'S\\u00ED',
  };

  // Subir foto si se seleccion\\u00F3
  const fotoInput = document.getElementById('fotoActorInput');
  if (fotoInput?.files[0]) {
    const file = fotoInput.files[0];
    const reader = new FileReader();
    reader.onload = e => {
      const base64 = e.target.result.split(',')[1];
      const mime   = file.type;
      const idTemp = 'ACT-' + Date.now();
      toast('\\u23F3 Subiendo foto...');
      google.script.run
        .withSuccessHandler(resFoto => {
          if (resFoto && resFoto.ok) {
            datos.Fotograf\\u00EDa = resFoto.url;
            toast('\\u2705 Foto subida');
          }
          guardarActorEnSheets(datos);
        })
        .withFailureHandler(()=>guardarActorEnSheets(datos))
        .procesarFoto(base64, mime, 'actores', idTemp, nombre);
    };
    reader.readAsDataURL(file);
  } else {
    guardarActorEnSheets(datos);
  }
}

function guardarActorEnSheets(datos) {
  google.script.run
    .withSuccessHandler(res=>{
      if(res.ok){
        toast('Actor guardado correctamente');
        CACHE.actores=null;
        cargarActores();
        sTab('actores','lista',document.querySelector('#actoresTabs .tb'));
      } else toast(res.msg,'err');
    })
    .withFailureHandler(()=>toast('Error guardando actor','err'))
    .guardarActor(datos);
}

function guardarNuevoSectorista() {
  var nombre = document.getElementById('nsec_nombre').value.trim();
  var dni    = document.getElementById('nsec_dni').value.trim();
  var clave  = document.getElementById('nsec_clave').value.trim();
  if (!nombre || !dni || !clave) { toast('Complete nombre, DNI y clave','err'); return; }
  var id = 'SEC-' + String(((CACHE.sectoristas||[]).length)+1).padStart(2,'0');
  var datos = {
    ID_Sectorista: id,
    Nombre_Sectorista: nombre, DNI: dni, clave_Sectorista: clave,
    Cargo: (document.getElementById('nsec_cargo')||{}).value || '',
    ID_Area: (document.getElementById('nsec_area')||{}).value || '',
    Email: (document.getElementById('nsec_email')||{}).value || '',
    Telefono: (document.getElementById('nsec_tel')||{}).value || '',
    Estado: 'Activo', N_Casos: 0
  };
  var fotoInput = document.querySelector('#sec-nuevo input[type=file]');
  if (fotoInput && fotoInput.files && fotoInput.files[0]) {
    var file = fotoInput.files[0];
    var reader = new FileReader();
    reader.onload = function(e) {
      toast('\\u23F3 Guardando...');
      google.script.run
        .withSuccessHandler(function(rf) {
          if (rf && rf.ok) datos['Fotograf\\u00EDa'] = rf.url;
          _guardarPersonaFinal(datos, 'guardarSectorista', function(){ CACHE.sectoristas=null; cargarSectoristas(); });
        })
        .withFailureHandler(function(){ _guardarPersonaFinal(datos, 'guardarSectorista', function(){ CACHE.sectoristas=null; cargarSectoristas(); }); })
        .guardarFotoPersona(e.target.result.split(',')[1], file.type, 'sectorista', id, nombre);
    };
    reader.readAsDataURL(file);
  } else {
    _guardarPersonaFinal(datos, 'guardarSectorista', function(){ CACHE.sectoristas=null; cargarSectoristas(); });
  }
}

function _guardarPersonaFinal(datos, fnName, callback) {
  google.script.run
    .withSuccessHandler(function(r){ if(r&&r.ok){toast('Guardado correctamente');if(callback)callback();}else toast((r&&r.msg)||'Error','err'); })
    .withFailureHandler(function(e){ toast('Error: '+(e.message||e),'err'); })
    [fnName](datos);
}

function xTab(btn, contentId, paneId) {
  var div = document.getElementById(contentId);
  if (!div) return;
  div.querySelectorAll('.tpane').forEach(function(t){ t.classList.remove('on'); });
  var p = document.getElementById(paneId);
  if (p) p.classList.add('on');
  if (btn && btn.closest) {
    var tabs = btn.closest('.tabs');
    if (tabs) tabs.querySelectorAll('.tb').forEach(function(b){ b.classList.remove('on'); });
    btn.classList.add('on');
  }
}

function prevFoto(input, previewId) {
  var file = input.files[0];

function verLineaTiempo(codigoCaso) {
  var modal = document.getElementById('modalDetalleInner');
  modal.innerHTML =
    '<div class="modal-hd"><div><h2>\\uD83D\\uDCC5 L\\u00EDnea de Tiempo</h2>' +
    '<div style="font-size:12px;color:var(--g500)">Caso: ' + codigoCaso + '</div></div>' +
    '<button class="mclose" onclick="cerrarModal()">\\u2715</button></div>' +
    '<div class="modal-bd" id="lineaTiempoContent">' +
    '<div class="spinner-wrap"><div class="spinner"></div><p>Cargando hechos...</p></div>' +
    '</div>' +
    '<div class="modal-ft"><button class="btn bs" onclick="cerrarModal()">Cerrar</button></div>';
  document.getElementById('modalDetalle').classList.remove('hidden');

  // Cargar hechos del caso
  var hechos = (CACHE.hechos||[]).filter(function(h){ return h.ID_Caso === codigoCaso; });
  if (hechos.length > 0) {
    renderLineaTiempo(hechos, codigoCaso);
  } else {
    google.script.run
      .withSuccessHandler(function(r) {
        if (r && r.ok) {
          CACHE.hechos = r.data;
          var hFiltrados = r.data.filter(function(h){ return h.ID_Caso === codigoCaso; });
          renderLineaTiempo(hFiltrados, codigoCaso);
        }
      })
      .withFailureHandler(function(){ renderLineaTiempo([], codigoCaso); })
      .getHechos();
  }
}

function renderLineaTiempo(hechos, codigoCaso) {
  var div = document.getElementById('lineaTiempoContent');
  if (!div) return;
  if (!hechos.length) {
    div.innerHTML = '<div style="text-align:center;padding:30px;color:var(--g500)">Sin hechos registrados para este caso</div>';
    return;
  }
  var items = hechos.sort(function(a,b){ return new Date(a.Fecha_Hecho||0) - new Date(b.Fecha_Hecho||0); })
    .map(function(h) {
      return '<div class="tl-item">' +
        '<div class="tl-date">' + fmtFecha(h.Fecha_Hecho) + '</div>' +
        '<div class="tl-type">' + (h.Tipo_Hecho||'Hecho') + '</div>' +
        '<div class="tl-desc">' + (h.Descripcion_Hecho||'') + '</div>' +
        '</div>';
    }).join('');
  div.innerHTML = '<div class="timeline">' + items + '</div>';
}

function renderBridgeLista(containerId, items, keyField, labelField, extraField, removeFn) {
  var div = document.getElementById(containerId);
  if (!div) return;
  if (!items || !items.length) {
    div.innerHTML = '<span style="font-size:12px;color:var(--g500)">Sin elementos agregados</span>';
    return;
  }
  // Usar DOM en lugar de innerHTML para evitar problemas de comillas
  div.innerHTML = '';
  items.forEach(function(item) {
    var label = item[labelField] || item[keyField] || '';
    var extra = extraField && item[extraField] ? ' (' + item[extraField] + ')' : '';
    var id    = item[keyField];
    var span  = document.createElement('span');
    span.style.cssText = 'display:inline-flex;align-items:center;gap:4px;background:var(--blue-pale);color:var(--blue);padding:4px 10px;border-radius:20px;font-size:12px;font-weight:500;margin:2px';
    span.textContent = label + extra;
    var btn = document.createElement('button');
    btn.textContent = '\\u00D7';
    btn.style.cssText = 'background:none;border:none;color:var(--blue);cursor:pointer;font-size:14px;padding:0 0 0 4px;line-height:1';
    btn.setAttribute('data-id', id);
    btn.onclick = function() { removeFn(this.getAttribute('data-id')); };
    span.appendChild(btn);
    div.appendChild(span);
  });
}

var _bridgeGen = {};

function iniciarBridgesGen(prefix) {
  _bridgeGen[prefix] = { deptos: [], actores: [], recursos: [] };
  ['deptos','actores','recursos'].forEach(function(k) {
    var el = document.getElementById(prefix + '_' + k + '_lista');
    if (el) el.innerHTML = '';
  });
  // Llenar selects
  llenarSelectConCache(prefix + '_depto_add',   'departamentos', 'ID_Ubigeo',  'Departamento',   'getDepartamentos');
  llenarSelectConCache(prefix + '_actor_add',   'actores',       'ID_Actor',   'Nombre_Actor',   'getActores');
  llenarSelectConCacheRecurso(prefix + '_recurso_add');
}

function llenarSelectConCache(selId, cacheKey, valKey, lblKey, fn) {
  if (CACHE[cacheKey] && CACHE[cacheKey].length) {
    llenarSelect(selId, CACHE[cacheKey], valKey, lblKey);
  } else {
    google.script.run
      .withSuccessHandler(function(r){ if(r&&r.ok){ CACHE[cacheKey]=r.data; llenarSelect(selId, r.data, valKey, lblKey); } })
      .withFailureHandler(function(){})
      [fn]();
  }
}

function llenarSelectConCacheRecurso(selId) {
  if (CACHE.recursos && CACHE.recursos.length) {
    var colR = detectarColRecurso(CACHE.recursos);
    llenarSelect(selId, CACHE.recursos, 'ID_Recurso', colR);
  } else {
    google.script.run
      .withSuccessHandler(function(r){
        if(r&&r.ok){
          CACHE.recursos = r.data;
          var colR = detectarColRecurso(r.data);
          llenarSelect(selId, r.data, 'ID_Recurso', colR);
        }
      })
      .withFailureHandler(function(){})
      .getRecursos();
  }
}

function detectarColRecurso(data) {
  if (!data || !data.length) return 'Nombre_comun_Recurso';
  var keys = Object.keys(data[0]);
  var found = keys.find(function(k){ return (k.indexOf('Nombre_c') >= 0 || k.indexOf('Nombre_com') >= 0) && k.toLowerCase().indexOf('recurso') >= 0; });
  return found || keys.find(function(k){ return k.toLowerCase().indexOf('nombre') >= 0; }) || 'Nombre_comun_Recurso';
}

function renderRecorrido(pasos, tipo) {
  var div = document.getElementById('recorridoContent');
  if (!div) return;
  if (!pasos || !pasos.length) {
    div.innerHTML = '<p style="color:var(--g500);text-align:center;padding:20px">Sin pasos registrados</p>';
    return;
  }

  var nombresDefault = tipo === 'caso'
    ? ['Elaboraci\\u00F3n','Revisi\\u00F3n Directora','Validado','En atenci\\u00F3n','Cierre']
    : ['Elaboraci\\u00F3n','Revisi\\u00F3n Directora','Validado','Remitido','Cierre'];

  var html = '<div class="pflow">';
  pasos.forEach(function(p, i) {
    var paso = p.Paso_caso || p.Paso_Num || (i+1);
    var nombre = nombresDefault[paso-1] || ('Paso ' + paso);
    var fechaFin = p.Fecha_Fin || p.Fecha_fin || '';
    var cerrado = !!fechaFin;
    var activo  = !cerrado && i === pasos.length - 1;
    var cls = cerrado ? 'done' : (activo ? 'active' : '');
    html +=
      '<div class="pstep ' + cls + '">' +
        '<div class="ps-c ' + cls + '">' + paso + '</div>' +
        '<div class="ps-l ' + cls + '">' + nombre +
          (cerrado ? '<br>' + fmtFecha(fechaFin) : '') +
          (activo  ? '<br>En curso' : '') +
        '</div>' +
      '</div>';
  });
  html += '</div>';

  // Tabla detalle
  html += '<div class="tw" style="margin-top:16px"><table>' +
    '<thead><tr><th>Paso</th><th>Responsable</th><th>Fecha inicio</th><th>Fecha fin</th><th>D\\u00EDas</th><th>Observaciones</th></tr></thead><tbody>';
  pasos.forEach(function(p) {
    var paso = p.Paso_caso || p.Paso_Num || '';
    html += '<tr>' +
      '<td><span class="badge b-blu">Paso ' + paso + '</span></td>' +
      '<td style="font-size:12px">' + (p.ID_Responsable||'—') + '</td>' +
      '<td style="font-size:12px">' + fmtFecha(p.Fecha_inicio||p.Fecha_Inicio||'') + '</td>' +
      '<td style="font-size:12px">' + fmtFecha(p.Fecha_Fin||p.Fecha_fin||'') + '</td>' +
      '<td style="font-size:12px">' + (p.Dias_Demora||0) + 'd</td>' +
      '<td style="font-size:12px;max-width:200px">' + (p.Observaciones||'—') + '</td>' +
    '</tr>';
  });
  html += '</tbody></table></div>';
  div.innerHTML = html;
}
